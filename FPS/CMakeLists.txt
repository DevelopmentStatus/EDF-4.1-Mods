cmake_minimum_required(VERSION 3.20)
project(EDF41FPS LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "EDF 4.1 requires x64")
endif()
include(CTest)
add_executable(fps_core_tests tests/core_tests.cpp)
target_include_directories(fps_core_tests PRIVATE src)
add_test(NAME fps_core COMMAND fps_core_tests)
add_test(NAME fps_head_materials COMMAND fps_core_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/head-materials.tsv")
if(WIN32)
  set(MH third_party/minhook)
  add_library(fps_minhook STATIC ${MH}/src/buffer.c ${MH}/src/hook.c ${MH}/src/trampoline.c ${MH}/src/hde/hde64.c)
  target_include_directories(fps_minhook PUBLIC ${MH}/include)
  add_executable(fps_hook_probe tests/hook_probe.cpp)
  target_link_libraries(fps_hook_probe PRIVATE fps_minhook)
  target_compile_definitions(fps_hook_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  add_library(EDF41FPS SHARED src/plugin.cpp)
  target_link_libraries(EDF41FPS PRIVATE fps_minhook bcrypt user32 xinput9_1_0)
  target_compile_definitions(EDF41FPS PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_compile_options(EDF41FPS PRIVATE /W4 /permissive- /utf-8)
  target_link_options(EDF41FPS PRIVATE /MAP)
  add_executable(fps_render_tests tests/render_tests.cpp)
  target_link_libraries(fps_render_tests PRIVATE fps_minhook bcrypt user32 xinput9_1_0)
  target_compile_definitions(fps_render_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_compile_options(fps_render_tests PRIVATE /utf-8)
  add_test(NAME fps_render_scope COMMAND fps_render_tests)
endif()
