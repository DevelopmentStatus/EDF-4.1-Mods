cmake_minimum_required(VERSION 3.20)
project(EDF41FPSVehicles LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "EDF 4.1 requires Windows x64")
endif()
include(CTest)
set(MH ../FPS/third_party/minhook)
add_library(vehicle_minhook STATIC ${MH}/src/buffer.c ${MH}/src/hook.c ${MH}/src/trampoline.c ${MH}/src/hde/hde64.c)
target_include_directories(vehicle_minhook PUBLIC ${MH}/include)
add_library(EDF41FPSVehicles SHARED src/plugin.cpp)
add_executable(vehicle_render_tests tests/render_tests.cpp)
add_executable(vehicle_core_tests ../FPS/tests/core_tests.cpp)
add_executable(vehicle_orientation_tests tests/orientation_tests.cpp)
target_include_directories(vehicle_core_tests PRIVATE src)
add_executable(vehicle_hook_probe ../FPS/tests/hook_probe.cpp)
add_executable(vehicle_fade_tests tests/fade_tests.cpp)
target_link_libraries(vehicle_fade_tests PRIVATE vehicle_minhook d3d11 d3dcompiler)
target_compile_definitions(vehicle_fade_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(vehicle_hook_probe PRIVATE vehicle_minhook)
foreach(target EDF41FPSVehicles vehicle_render_tests)
  target_link_libraries(${target} PRIVATE vehicle_minhook bcrypt user32 xinput9_1_0 d3d11)
  target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
endforeach()
target_link_options(EDF41FPSVehicles PRIVATE /MAP)
add_test(NAME vehicle_core COMMAND vehicle_core_tests)
add_test(NAME vehicle_head_materials COMMAND vehicle_core_tests "${CMAKE_CURRENT_SOURCE_DIR}/../FPS/tests/head-materials.tsv")
add_test(NAME vehicle_render COMMAND vehicle_render_tests)
add_test(NAME vehicle_fade_pixels COMMAND vehicle_fade_tests)
add_test(NAME vehicle_orientation COMMAND vehicle_orientation_tests)
